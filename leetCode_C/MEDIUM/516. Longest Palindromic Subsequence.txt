int longestPalindromeSubseq(char* s) {
    int n = strlen(s);
    char s1[n + 1];
    for (int i = 0; i < n; i++) {
        s1[i] = s[n - i - 1];
    }
    s1[n] = '\0';
    int L[n + 1][n + 1];
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= n; j++) {
            if (i == 0 || j == 0) L[i][j] = 0;
            else if (s[i - 1] == s1[j - 1]) L[i][j] = 1 + L[i - 1][j - 1];
            else L[i][j] = (L[i - 1][j] > L[i][j - 1]) ? L[i - 1][j] : L[i][j - 1];
        }
    }
    int length = L[n][n];
    return length;
}
