int getDistance(int* p1, int* p2) {
    return abs(p1[0] - p2[0]) + abs(p1[1] - p2[1]);
}

int minCostConnectPoints(int** points, int pointsSize, int* pointsColSize) {
    if (pointsSize <= 1) return 0;

    int *minDist = (int *)malloc(pointsSize * sizeof(int));
    int *inMST = (int *)calloc(pointsSize, sizeof(int));

    for (int i = 0; i < pointsSize; i++) {
        minDist[i] = getDistance(points[0], points[i]);
    }

    inMST[0] = 1;
    int totalCost = 0;
    int edgeConnected = 0;

    while (edgeConnected < pointsSize - 1) {
        int nextnode = -1;
        int min = 9999999;
        for (int i = 0; i < pointsSize; i++) {
            if (!inMST[i] && minDist[i] < min) {
                min = minDist[i];
                nextnode = i;
            }
        }

        inMST[nextnode] = 1;
        totalCost += min;
        edgeConnected++;

        for (int i = 0; i < pointsSize; i++) {
            if (!inMST[i]) {
                int d = getDistance(points[nextnode], points[i]);
                if (d < minDist[i]) {
                    minDist[i] = d;
                }
            }
        }
    }
    free(minDist);
    free(inMST);
    return totalCost;
}
